% arch between legs a and b (above the fibre); legs are small rings
\newcommand{\arch}[3]{\pgfmathsetmacro{\hh}{0.42*(#2-#1)+0.55}\draw[#3] (#1,0.08) .. controls (#1,\hh) and (#2,\hh) .. (#2,0.08);}
\newcommand{\leg}[1]{\legdot{#1}{0}{cohesin}}
\newcommand{\fibre}{\draw[fiber] (0.2,0)--(7.0,0);}
% motion of a leg (it sits on the fibre; the arrow hovers above it): free = arrowhead, stopped = blunt bar in the colour of the CTCF
\newcommand{\blkL}[2]{\draw[cohesin,line width=1.8pt,line cap=round] (#1,-0.85)--(#1-0.45,-0.85);\draw[#2,line width=3.4pt,line cap=round] (#1-0.52,-1.05)--(#1-0.52,-0.65);}
\newcommand{\blkR}[2]{\draw[cohesin,line width=1.8pt,line cap=round] (#1,-0.85)--(#1+0.45,-0.85);\draw[#2,line width=3.4pt,line cap=round] (#1+0.52,-1.05)--(#1+0.52,-0.65);}
\newcommand{\goL}[1]{\draw[cohesin,line width=1.8pt,-{Stealth[length=7pt,width=8pt]},line cap=round] (#1,-0.85)--(#1-0.8,-0.85);}
\newcommand{\goR}[1]{\draw[cohesin,line width=1.8pt,-{Stealth[length=7pt,width=8pt]},line cap=round] (#1,-0.85)--(#1+0.8,-0.85);}
